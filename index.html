SQL Showcase

Where the logic lives

Two patterns I use often at Anubavam — reconciliation and incremental loading.

reconciliation.sqlData Quality
-- Source-to-target reconciliation check
WITH src AS (
  SELECT student_id, COUNT(*) AS src_cnt
  FROM staging.enrollment_raw
  GROUP BY student_id
),
tgt AS (
  SELECT student_id, COUNT(*) AS tgt_cnt
  FROM datahub.fact_enrollment
  GROUP BY student_id
)
SELECT s.student_id, s.src_cnt, t.tgt_cnt
FROM src s
LEFT JOIN tgt t ON s.student_id = t.student_id
WHERE s.src_cnt IS DISTINCT FROM t.tgt_cnt;
incremental_load.sqlETL Pattern
-- Timestamp-based incremental upsert
MERGE INTO datahub.dim_faculty AS tgt
USING (
  SELECT *
  FROM staging.faculty_raw
  WHERE updated_at > '{{ last_watermark }}'
) AS src
ON tgt.faculty_id = src.faculty_id
WHEN MATCHED THEN UPDATE SET
  tgt.department = src.department,
  tgt.updated_at = src.updated_at
WHEN NOT MATCHED THEN INSERT
  (faculty_id, department, updated_at)
VALUES (src.faculty_id, src.department, src.updated_at);
